String getMacAddress() {
  uint64_t chipid = ESP.getEfuseMac();
  char macStr[18];
  // จัดรูปแบบให้เป็น AA:BB:CC:DD:EE:FF
  snprintf(macStr, sizeof(macStr), "%02X:%02X:%02X:%02X:%02X:%02X",
           (uint8_t)(chipid >> 0), (uint8_t)(chipid >> 8),
           (uint8_t)(chipid >> 16), (uint8_t)(chipid >> 24),
           (uint8_t)(chipid >> 32), (uint8_t)(chipid >> 40));
  return String(macStr);
}

Point sensor("room_sensor1"); 

void sendData(String macID,const char* wifi, float temp, float dhttemp, float humid,float angle) { 

    Serial.print("ESP32 MAC Address: "); 
    Serial.print(macID); 
    Serial.print("ESP32 Wifi Name: "); 
    Serial.print(wifi); 
    Serial.print("ESP32 Tempp: "); 
    Serial.print(temp); 
    Serial.print(" *C\t"); 
    Serial.print("DHT22.Tempp: "); 
    Serial.print(dhttemp); 
    Serial.print(" *C\t"); 
    Serial.print("Humidity: "); 
    Serial.print(humid); 
    Serial.print(" %\t"); 
    Serial.print("Rotary Angle: "); 
    Serial.print(angle); 
    Serial.println(" deg"); 

    sensor.clearFields();
    sensor.clearTags();

    sensor.addTag("ESP32-MAC-Address", macID); 
    sensor.addTag("ESP32-Wifi-Name", wifi); 
    sensor.addField("MAC_ID", macID.c_str()); 
    sensor.addField("WIFI_NAME", wifi);
    sensor.addField("ESP32-Tempp", temp); 
    sensor.addField("DHT22-Tempp", dhttemp); 
    sensor.addField("humidity", humid); 
    sensor.addField("rotary_angle", angle); 
    Serial.print("Writing: "); 
    Serial.println(client.pointToLineProtocol(sensor)); 

    if (!client.writePoint(sensor)) { 
        Serial.print("InfluxDB write failed: "); 
        Serial.println(client.getLastErrorMessage()); 
    } 
} 

void setup() { 
    Serial.begin(115200); 
    pinMode(CLK_PIN, INPUT);
    pinMode(DT_PIN, INPUT);

    dht.setup(Pin_DHT22, DHTesp::DHT22);

    Serial.println(ssid); 

    WiFi.begin(ssid, password); 
    while (WiFi.status() != WL_CONNECTED) { 
        delay(500); 
        Serial.print("."); 
    } 

    Serial.println(""); 
    Serial.println("WiFi connected"); 
    Serial.println("IP address: "); 
    Serial.println(WiFi.localIP()); 

    if (client.validateConnection()) { 
        Serial.print("Connected to InfluxDB: "); 
        Serial.println(client.getServerUrl()); 
    } else { 
        Serial.print("InfluxDB connection failed: "); 
        Serial.println(client.getLastErrorMessage()); 
    }
}

        if (counter > 0) {
            counter = counter - 36 ;
        } else if (counter == 0) {
            counter = 360 ;
