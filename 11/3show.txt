from(bucket: "kit")
  |> range(start: v.timeRangeStart, stop: v.timeRangeStop)
  |> filter(fn: (r) => r["_measurement"] == "room_sensor1")
  |> pivot(rowKey:["_time"], columnKey: ["_field"], valueColumn: "_value")
  |> keep(columns: ["_time", "MAC_ID", "WIFI_NAME", "ESP32-Tempp", "DHT22-Tempp", "humidity", "rotary_angle"])
  |> sort(columns: ["_time"], desc: true)
