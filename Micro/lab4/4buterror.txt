// ฟังก์ชันส่ง HTTP Request โดยรับพารามิเตอร์เป็นข้อความ 
void sendHttpRequest(String message) {
  client.setInsecure();
  String url = "https://api.telegram.org/bot" + telegramToken 
               + "/sendMessage?chat_id=" + chatID 
               + "&text=" + urlEncode(message);
  
  http.begin(client, url);
  int httpCode = http.GET();
  if (httpCode > 0) {
    Serial.println(http.getString());
  } else {
    Serial.printf("Error: %s\n", http.errorToString(httpCode).c_str());
  }
  http.end();
}
 
 
// ฟังก์ชันสําหรับแปลงข้อความเป็น URL Encoding (สําหรับรองรับการใช้หลายคํา) 
String urlEncode(String message) { 
  String encoded = ""; 
  char c; 
  for (int i = 0; i < message.length(); i++) { 
    c = message[i]; 
    if (c == ' ') { 
      encoded += "%20";   // เปลียนช่องว่างเป็น %20 
    } else if (c == '&') { 
      encoded += "%26";   // เปลียนเครืองหมาย & เป็น %26 
    } else if (c == '=') { 
      encoded += "%3D";   // เปลียนเครืองหมาย = เป็น %3D 
    } else { 
      encoded += c;   // ตัวอักษรอืนๆ ไม่ต้องแปลง 
    } 
  } 
  return encoded; 
}
