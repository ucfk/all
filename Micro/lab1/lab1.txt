void loop() {
  // อ่านค่าจากปุ่มมาเก็บในตัวแปร buttonState
  int buttonState = digitalRead(pushButton); 
  Serial.println(buttonState); // ดูสถานะในจอ Monitor

  if (buttonState == LOW) { // 0 = เหยียบเบรก (กดแช่)
    
    // --- สเต็ป 1: กะพริบ 10 ครั้ง ---
    for (int i = 0; i < 10; i++) {
      // เช็คระหว่างทาง: ถ้าปล่อยเบรกกลางคัน ให้เลิกกะพริบและออกจาก loop นี้ทันที
      if (digitalRead(pushButton) == HIGH) break; 
      
      allon();   delay(100);
      off();     delay(100);
    }

    // --- สเต็ป 2: หลังจากกะพริบครบ ถ้ายังเหยียบแช่อยู่ ให้ติดค้าง ---
    while (digitalRead(pushButton) == LOW) {
      allon(); 
    }
    
  } else { // 1 = ปล่อยเบรก
    off(); // ดับไฟทันที
  }
}
