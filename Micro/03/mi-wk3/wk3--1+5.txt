ORG  0000H  
    MOV  DPTR, #9000H ; เริ่มต้นที่ Address 9000H
    MOV  A, #0AAH     ; เตรียมค่า AA ที่จะเติม
    MOV  R2, #96      ; ลูปนอก (Outer Loop): ทำทั้งหมด 96 ชุด

OUTER_LOOP:
    MOV  R1, #0       ; ลูปใน (Inner Loop): เริ่มที่ 0 เพื่อให้นับได้ 256 รอบ

INNER_LOOP:
    MOVX @DPTR, A     ; เขียนค่า AA ลงใน External RAM
    INC  DPTR         ; ขยับตำแหน่ง Address ไปทีละ 1
    DJNZ R1, INNER_LOOP ; ลดค่า R1 ถ้ายังไม่เป็น 0 ให้ทำในลูปในต่อ (256 รอบ)

    DJNZ R2, OUTER_LOOP ; เมื่อครบ 256 รอบ ให้ลดค่า R2 (ชุดที่ 96 -> 95)
                        ; ถ้ายังไม่ครบ 96 ชุด ให้กลับไปเริ่มลูปในใหม่

    JMP  $            ; จบการทำงาน วนอยู่ที่เดิม
    END
