ORG  0000H  
    MOV  DPTR, #1000H ; ตัวชี้ต้นทาง (External)
    MOV  R0, #40H     ; ตัวชี้ปลายทาง (Internal)

LOOP:  
    MOVX A, @DPTR     ; อ่านข้อมูลจาก External RAM มาไว้ใน A
    MOV  @R0, A       ; เขียนข้อมูลจาก A ลงใน Internal RAM
    INC  DPTR         ; เลื่อนตำแหน่งต้นทาง
    INC  R0           ; เลื่อนตำแหน่งปลายทาง
    
    ; เช็คว่าทำครบหรือยัง (เราต้องการทำถึง 101FH ดังนั้นตัวถัดไปคือ 1020H)
    ; แต่ DPTR เป็น 16 บิต เช็คยาก เราจึงเช็คที่ R0 แทนจะง่ายกว่า
    ; เริ่มที่ 40H ทำ 32 ครั้ง (20H) ดังนั้นตัวสุดท้ายคือ 5FH ตัวถัดไปคือ 60H
    MOV  A, R0  
    CJNE A, #60H, LOOP ; ถ้ายังไม่ถึง 60H ให้กลับไปทำต่อ
    
    JMP  $  
    END
