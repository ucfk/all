import machine
import dht
import network
import socket
import tm1638
import time

# --- ตั้งค่า Wi-Fi ---
SSID = 'Cgrs'
PASSWORD = '@@@@@@@@'

# --- ตั้งค่าอุปกรณ์ ---
tm = tm1638.TM1638(stb=machine.Pin(2), clk=machine.Pin(4), dio=machine.Pin(5))
sensor = dht.DHT22(machine.Pin(15))

# ตัวแปรสถานะ
temp = 0.0
hum = 0.0
last_read = 0

def connect_wifi():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(False)
    time.sleep(0.5)
    wlan.active(True)
    wlan.connect(SSID, PASSWORD)
    
    tm.show('Con WiFi')
    print('Connecting to WiFi...', end='')
    
    timeout = 20
    while not wlan.isconnected() and timeout > 0:
        print('.', end='')
        time.sleep(1)
        timeout -= 1
        
    if wlan.isconnected():
        ip = wlan.ifconfig()[0]
        print('\nConnected! IP:', ip)
        tm.show('IP ' + ip.split('.')[-1])
        time.sleep(2)
        return ip
    else:
        print('\nFailed!')
        tm.show('ERR WIFI')
        return None

def web_page(t, h):
    html = f"""

# เริ่มระบบ
ip = connect_wifi()
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1) # แก้ปัญหา EADDRINUSE
s.bind(('', 80))
s.listen(5)

print("Server is running...")

while True:
    try:
        # 1. อ่านค่า Sensor ทุก 2 วินาที (ไม่ใช้ sleep)
        if time.ticks_diff(time.ticks_ms(), last_read) > 2000:
            sensor.measure()
            temp = sensor.temperature()
            hum = sensor.humidity()
            
            # แสดงผลที่จอ TM1638: "24.4C33.6h"
            tm.show("{:2.1f}C{:2.1f}h".format(temp, hum))
            last_read = time.ticks_ms()

        # 2. ตรวจสอบการเรียกหน้าเว็บ (Non-blocking)
        s.settimeout(0.1)
        try:
            conn, addr = s.accept()
            request = conn.recv(1024)
            response = web_page(temp, hum)
            conn.send('HTTP/1.1 200 OK\nContent-Type: text/html\nConnection: close\n\n')
            conn.sendall(response)
            conn.close()
        except OSError:
            pass # ไม่มีคนเข้าเว็บก็ข้ามไป

    except Exception as e:
        print("Error:", e)
