int pushButton = 15;  
int led[] = { 23, 22, 21, 19, 18, 5, 4, 2 }; 
volatile int lastled=0; 
void onfromR(int last){ 
    if(last==0){last=7;} 
   for(int i=last;i>=0;i--){ 
   digitalWrite(led[i], HIGH); 
   delay(200);  
   digitalWrite(led[i], LOW); 
   if(digitalRead(pushButton)!=0){ 
    lastled=i; 
    break; 
    }else if(i==0){lastled=0;} 
   }  
}
